cmake_minimum_required(VERSION 3.20)
project(save_lock_lab LANGUAGES CXX)
set(GODOTCPP_API_VERSION "4.7" CACHE STRING "" FORCE)
set(GODOTCPP_TARGET "template_debug" CACHE STRING "" FORCE)
add_subdirectory("${GODOT_CPP_DIR}" godot-cpp)
add_library(save_lock_lab SHARED godot_lock.cpp)
target_link_libraries(save_lock_lab PRIVATE godot-cpp)
target_compile_features(save_lock_lab PRIVATE cxx_std_17)
set_target_properties(save_lock_lab PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "$<1:${CMAKE_CURRENT_SOURCE_DIR}/project/bin>" RUNTIME_OUTPUT_DIRECTORY "$<1:${CMAKE_CURRENT_SOURCE_DIR}/project/bin>")
